Code Reference for BackendUse this logic for the service file:const { google } = require('googleapis');
// Assumes you have setup authentication in a separate config
const chat = google.chat({ version: 'v1', auth: yourAuthClient });

// 1. Create a dedicated space for a specific trip
async function createTransportSpace(routeName) {
  try {
    const res = await chat.spaces.create({
      requestBody: {
        displayName: `🚐 Transport: ${routeName}`,
        spaceType: 'SPACE',
      },
    });
    return { 
      spaceId: res.data.name, 
      spaceUrl: `https://mail.google.com/chat/u/0/#chat/space/${res.data.name.split('/')[1]}`
    };
  } catch (err) {
    console.error("Failed to create Chat Space:", err);
    return null;
  }
}

// 2. Broadcast an "SOS" card to the regional network
async function broadcastAlert(rescueName, animalCount, notes, link) {
  const webhookUrl = process.env.REGIONAL_RESCUE_WEBHOOK_URL;
  
  const cardPayload = {
    cards: [{
      header: {
        title: "🚨 URGENT: Transfer Request",
        subtitle: `${rescueName} is over capacity`,
        imageUrl: "[https://irescue.life/assets/alert-icon.png](https://irescue.life/assets/alert-icon.png)"
      },
      sections: [{
        widgets: [
          {
            keyValue: {
              topLabel: "Animals Needing Transfer",
              content: `${animalCount} Animals`,
              icon: "DESCRIPTION"
            }
          },
          {
            textParagraph: { text: notes }
          },
          {
            buttons: [{
              textButton: {
                text: "View Animals & Help",
                onClick: {
                  openLink: { url: link }
                }
              }
            }]
          }
        ]
      }]
    }]
  };

  await fetch(webhookUrl, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(cardPayload)
  });
}

module.exports = { createTransportSpace, broadcastAlert };
