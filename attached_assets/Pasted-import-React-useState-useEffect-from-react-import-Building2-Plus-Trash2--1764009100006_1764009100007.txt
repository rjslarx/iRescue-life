import React, { useState, useEffect } from 'react';
import { 
  Building2, 
  Plus, 
  Trash2, 
  LayoutGrid, 
  Save, 
  Dog, 
  Home, 
  Edit2, 
  Check, 
  X,
  AlertCircle
} from 'lucide-react';

export default function KennelLayoutBuilder() {
  // --- State Management ---
  
  // This is the core data structure we discussed
  const [buildings, setBuildings] = useState([
    {
      id: 'b1',
      name: 'Main Shelter',
      rows: [
        { id: 'r1', name: 'Row A (Large Dogs)', capacity: 10, type: 'standard' },
        { id: 'r2', name: 'Row B (Small Dogs)', capacity: 8, type: 'standard' }
      ]
    }
  ]);

  const [activeBuildingId, setActiveBuildingId] = useState('b1');
  const [viewMode, setViewMode] = useState('edit'); // 'edit' or 'visualize'
  
  // Occupancy state (mapping kennel IDs to dog names/statuses)
  // Key format: buildingId-rowId-index
  const [occupancy, setOccupancy] = useState({});

  // --- Actions ---

  const addBuilding = () => {
    const newId = `b${Date.now()}`;
    const newBuilding = {
      id: newId,
      name: 'New Building',
      rows: []
    };
    setBuildings([...buildings, newBuilding]);
    setActiveBuildingId(newId);
  };

  const deleteBuilding = (id) => {
    if (buildings.length === 1) {
      alert("You must have at least one building.");
      return;
    }
    const newBuildings = buildings.filter(b => b.id !== id);
    setBuildings(newBuildings);
    if (activeBuildingId === id) setActiveBuildingId(newBuildings[0].id);
  };

  const updateBuildingName = (id, newName) => {
    setBuildings(buildings.map(b => b.id === id ? { ...b, name: newName } : b));
  };

  const addRow = (buildingId) => {
    setBuildings(buildings.map(b => {
      if (b.id !== buildingId) return b;
      return {
        ...b,
        rows: [...b.rows, { 
          id: `r${Date.now()}`, 
          name: 'New Row', 
          capacity: 5,
          type: 'standard' 
        }]
      };
    }));
  };

  const updateRow = (buildingId, rowId, field, value) => {
    setBuildings(buildings.map(b => {
      if (b.id !== buildingId) return b;
      return {
        ...b,
        rows: b.rows.map(r => r.id === rowId ? { ...r, [field]: value } : r)
      };
    }));
  };

  const deleteRow = (buildingId, rowId) => {
    setBuildings(buildings.map(b => {
      if (b.id !== buildingId) return b;
      return {
        ...b,
        rows: b.rows.filter(r => r.id !== rowId)
      };
    }));
  };

  const toggleOccupancy = (kennelKey) => {
    setOccupancy(prev => {
      const newState = { ...prev };
      if (newState[kennelKey]) {
        delete newState[kennelKey];
      } else {
        newState[kennelKey] = { status: 'occupied', name: 'Unknown Dog' };
      }
      return newState;
    });
  };

  // --- Derived State ---
  const activeBuilding = buildings.find(b => b.id === activeBuildingId) || buildings[0];

  return (
    <div className="flex flex-col h-screen bg-slate-50 text-slate-800 font-sans">
      
      {/* Header */}
      <header className="bg-white border-b border-slate-200 px-6 py-4 flex items-center justify-between shadow-sm z-10">
        <div className="flex items-center gap-3">
          <div className="bg-indigo-600 p-2 rounded-lg">
            <Home className="w-6 h-6 text-white" />
          </div>
          <div>
            <h1 className="text-xl font-bold text-slate-900">irescue.life</h1>
            <p className="text-xs text-slate-500 font-medium tracking-wide uppercase">Facility Manager</p>
          </div>
        </div>
        
        <div className="flex gap-2">
           <button 
            onClick={() => setViewMode('edit')}
            className={`px-4 py-2 rounded-md text-sm font-medium transition-colors ${viewMode === 'edit' ? 'bg-indigo-100 text-indigo-700' : 'text-slate-600 hover:bg-slate-100'}`}
          >
            <Edit2 className="w-4 h-4 inline mr-2" />
            Edit Layout
          </button>
          <button 
            onClick={() => setViewMode('visualize')}
            className={`px-4 py-2 rounded-md text-sm font-medium transition-colors ${viewMode === 'visualize' ? 'bg-indigo-100 text-indigo-700' : 'text-slate-600 hover:bg-slate-100'}`}
          >
            <LayoutGrid className="w-4 h-4 inline mr-2" />
            Visualizer
          </button>
          <button className="bg-indigo-600 hover:bg-indigo-700 text-white px-4 py-2 rounded-md text-sm font-medium flex items-center gap-2 shadow-sm transition-all ml-4">
            <Save className="w-4 h-4" />
            Save Changes
          </button>
        </div>
      </header>

      <div className="flex flex-1 overflow-hidden">
        
        {/* Sidebar: Building Navigation */}
        <aside className="w-64 bg-white border-r border-slate-200 flex flex-col">
          <div className="p-4 border-b border-slate-100">
            <h2 className="text-sm font-semibold text-slate-400 uppercase tracking-wider mb-3">Buildings</h2>
            <div className="space-y-2">
              {buildings.map(b => (
                <button
                  key={b.id}
                  onClick={() => setActiveBuildingId(b.id)}
                  className={`w-full text-left px-3 py-2 rounded-md text-sm flex items-center justify-between group transition-all ${
                    activeBuildingId === b.id 
                      ? 'bg-indigo-50 text-indigo-700 font-medium ring-1 ring-indigo-200' 
                      : 'text-slate-600 hover:bg-slate-50'
                  }`}
                >
                  <span className="flex items-center gap-2 truncate">
                    <Building2 className="w-4 h-4 opacity-70" />
                    {b.name}
                  </span>
                  {activeBuildingId === b.id && (
                     <span className="bg-indigo-200 text-indigo-800 text-[10px] px-1.5 py-0.5 rounded-full">
                       {b.rows.length} rows
                     </span>
                  )}
                </button>
              ))}
            </div>
            <button 
              onClick={addBuilding}
              className="mt-3 w-full border border-dashed border-slate-300 text-slate-500 hover:border-indigo-400 hover:text-indigo-600 p-2 rounded-md text-sm flex items-center justify-center gap-2 transition-colors"
            >
              <Plus className="w-4 h-4" />
              Add Building
            </button>
          </div>

          <div className="p-4 mt-auto bg-slate-50 border-t border-slate-200">
             <div className="text-xs text-slate-500 leading-relaxed">
               <strong className="text-slate-700">Tip:</strong> Create separate buildings for Quarantine or Intake areas to better manage disease control.
             </div>
          </div>
        </aside>

        {/* Main Content Area */}
        <main className="flex-1 overflow-y-auto bg-slate-50/50 p-8">
          
          <div className="max-w-5xl mx-auto">
            
            {/* Building Header config */}
            <div className="mb-8 flex items-end gap-4 border-b border-slate-200 pb-6">
              <div className="flex-1">
                <label className="block text-xs font-semibold text-slate-500 uppercase mb-1">Building Name</label>
                <input 
                  type="text" 
                  value={activeBuilding.name}
                  onChange={(e) => updateBuildingName(activeBuilding.id, e.target.value)}
                  className="text-3xl font-bold text-slate-800 bg-transparent border-none p-0 focus:ring-0 placeholder-slate-300 w-full"
                  placeholder="e.g. Main Kennel"
                />
              </div>
              <button 
                onClick={() => deleteBuilding(activeBuilding.id)}
                className="text-rose-500 hover:bg-rose-50 px-3 py-2 rounded-md text-sm font-medium transition-colors"
              >
                Delete Building
              </button>
            </div>

            {/* CONTENT SWITCHER */}
            {viewMode === 'edit' ? (
              
              /* --- EDIT MODE --- */
              <div className="space-y-6">
                <div className="flex items-center justify-between mb-2">
                  <h3 className="text-lg font-semibold text-slate-800">Row Configuration</h3>
                  <button 
                    onClick={() => addRow(activeBuilding.id)}
                    className="bg-white border border-slate-300 hover:bg-slate-50 text-slate-700 px-4 py-2 rounded-md text-sm font-medium shadow-sm transition-all flex items-center gap-2"
                  >
                    <Plus className="w-4 h-4" />
                    Add Row
                  </button>
                </div>

                {activeBuilding.rows.length === 0 ? (
                  <div className="text-center py-12 bg-white rounded-xl border border-dashed border-slate-300">
                    <div className="w-12 h-12 bg-slate-100 rounded-full flex items-center justify-center mx-auto mb-3">
                      <LayoutGrid className="w-6 h-6 text-slate-400" />
                    </div>
                    <h3 className="text-slate-900 font-medium">No rows configured</h3>
                    <p className="text-slate-500 text-sm mt-1">Start by adding a row of kennels to this building.</p>
                  </div>
                ) : (
                  <div className="grid gap-4">
                    {activeBuilding.rows.map((row, index) => (
                      <div key={row.id} className="bg-white p-4 rounded-xl border border-slate-200 shadow-sm flex items-center gap-4 group hover:border-indigo-300 transition-colors">
                        <div className="w-8 h-8 rounded-full bg-slate-100 flex items-center justify-center text-slate-500 font-bold text-sm">
                          {index + 1}
                        </div>
                        
                        <div className="flex-1 grid grid-cols-1 md:grid-cols-3 gap-4">
                          <div>
                            <label className="block text-xs font-medium text-slate-500 mb-1">Row Name</label>
                            <input 
                              type="text" 
                              value={row.name}
                              onChange={(e) => updateRow(activeBuilding.id, row.id, 'name', e.target.value)}
                              className="w-full text-sm border-slate-300 rounded-md focus:border-indigo-500 focus:ring-indigo-500"
                              placeholder="e.g. Row A"
                            />
                          </div>
                          <div>
                            <label className="block text-xs font-medium text-slate-500 mb-1">Capacity (Kennels)</label>
                            <input 
                              type="number" 
                              min="1"
                              max="100"
                              value={row.capacity}
                              onChange={(e) => updateRow(activeBuilding.id, row.id, 'capacity', parseInt(e.target.value) || 0)}
                              className="w-full text-sm border-slate-300 rounded-md focus:border-indigo-500 focus:ring-indigo-500"
                            />
                          </div>
                           <div>
                            <label className="block text-xs font-medium text-slate-500 mb-1">Type</label>
                            <select 
                              value={row.type}
                              onChange={(e) => updateRow(activeBuilding.id, row.id, 'type', e.target.value)}
                              className="w-full text-sm border-slate-300 rounded-md focus:border-indigo-500 focus:ring-indigo-500"
                            >
                              <option value="standard">Standard Run</option>
                              <option value="isolation">Isolation/Quarantine</option>
                              <option value="puppy">Puppy Pen</option>
                              <option value="cat">Cat Condo</option>
                            </select>
                          </div>
                        </div>

                        <button 
                          onClick={() => deleteRow(activeBuilding.id, row.id)}
                          className="p-2 text-slate-400 hover:text-rose-500 hover:bg-rose-50 rounded-md transition-colors opacity-0 group-hover:opacity-100"
                          title="Delete Row"
                        >
                          <Trash2 className="w-5 h-5" />
                        </button>
                      </div>
                    ))}
                  </div>
                )}
              </div>

            ) : (
              
              /* --- VISUALIZE MODE --- */
              <div className="space-y-8 animate-in fade-in duration-300">
                <div className="bg-indigo-900 text-white p-4 rounded-lg flex items-start gap-3 shadow-lg">
                  <AlertCircle className="w-5 h-5 text-indigo-300 flex-shrink-0 mt-0.5" />
                  <div>
                    <h4 className="font-semibold text-sm">Interactive Map Preview</h4>
                    <p className="text-xs text-indigo-200 mt-1">
                      This is how your staff will see the shelter. Click any kennel box to toggle its occupancy status for testing purposes.
                    </p>
                  </div>
                </div>

                {activeBuilding.rows.map((row) => (
                  <div key={row.id} className="bg-white rounded-xl border border-slate-200 shadow-sm overflow-hidden">
                    <div className="bg-slate-50 border-b border-slate-200 px-4 py-3 flex justify-between items-center">
                      <h3 className="font-semibold text-slate-700">{row.name}</h3>
                      <span className="text-xs font-medium px-2 py-1 bg-white border border-slate-200 rounded-md text-slate-500 uppercase">
                        {row.type} • {row.capacity} Units
                      </span>
                    </div>
                    
                    <div className="p-6">
                      {/* The Grid: 
                          We use flex-wrap here, but you could calculate grid-cols based on capacity if you want strict rows.
                          Flex wrap handles varying screen sizes better for this use case.
                      */}
                      <div className="flex flex-wrap gap-3">
                        {Array.from({ length: row.capacity }).map((_, i) => {
                          const kennelKey = `${activeBuilding.id}-${row.id}-${i}`;
                          const isOccupied = occupancy[kennelKey];
                          
                          return (
                            <button
                              key={i}
                              onClick={() => toggleOccupancy(kennelKey)}
                              className={`
                                relative w-20 h-24 rounded-lg border-2 flex flex-col items-center justify-center gap-1 transition-all
                                ${isOccupied 
                                  ? 'bg-indigo-50 border-indigo-500 shadow-indigo-100' 
                                  : 'bg-white border-slate-200 hover:border-slate-400 hover:shadow-md'
                                }
                              `}
                            >
                              <span className={`text-xs font-bold ${isOccupied ? 'text-indigo-600' : 'text-slate-400'}`}>
                                #{i + 1}
                              </span>
                              
                              {isOccupied ? (
                                <>
                                  <Dog className="w-6 h-6 text-indigo-600" />
                                  <div className="absolute top-1 right-1 w-2 h-2 bg-indigo-500 rounded-full animate-pulse" />
                                </>
                              ) : (
                                <div className="w-6 h-6 rounded-full bg-slate-100 flex items-center justify-center">
                                  <Plus className="w-3 h-3 text-slate-400" />
                                </div>
                              )}
                            </button>
                          );
                        })}
                      </div>
                    </div>
                  </div>
                ))}

                {activeBuilding.rows.length === 0 && (
                   <p className="text-center text-slate-400 italic">No layout configured for this building yet.</p>
                )}
              </div>
            )}
            
          </div>
        </main>
      </div>
    </div>
  );
}
